#include <tunables/global>

# AppArmor profile for the Dashy app. The Supervisor loads it instead of
# Docker's default profile and renames the profile on the next line to the
# app's full slug, such as local_dashy or 1a2b3c4d_dashy; the signal rules
# match that name with *_dashy. Only the top profile may start at the
# beginning of a line: the Supervisor refuses a file with more than one.
#
# The top profile covers tini, run.sh and the tools it runs as root, and
# nginx. Node.js, so Dashy, the login check and the health check, runs in the
# child profile "node", which can read its own files, write only to the app
# configuration folder and /tmp, and run nothing but ping.
#
# .docs/development.md explains how to test a change to this file.
profile dashy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # run.sh copies the starter dashboard into /config and gives the folder to
  # the node user, starts Dashy and the login check as node with su-exec, and
  # stops them; the nginx master starts its worker as nginx.
  capability chown,
  capability dac_override,
  capability kill,
  capability setgid,
  capability setuid,

  network inet stream,
  network inet6 stream,
  network unix stream,

  # Docker sends stop signals from outside the profile: unconfined, or from
  # runc's own profile on hosts that have one.
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,
  signal (send, receive) peer=*_dashy,
  signal (send) peer=*_dashy//node,

  # Programs
  /sbin/tini ix,
  /run.sh rix,
  /bin/bash rix,
  /bin/busybox ix,
  /usr/bin/jq ix,
  /sbin/su-exec ix,
  /usr/sbin/nginx ix,
  /usr/local/bin/node cx -> node,

  # Docker starts the health check, node /app/services/healthcheck.js, in this
  # profile directly, so it does not switch to the node profile.
  /usr/local/bin/node rm,
  /app/package.json r,
  /app/services/healthcheck.js r,
  /sys/fs/cgroup/** r,
  @{PROC}/[0-9]*/cgroup r,
  @{PROC}/version_signature r,

  # System files and libraries
  /etc/** r,
  /lib/** mr,
  /usr/lib/** mr,
  /usr/share/** r,
  /dev/tty rw,

  # The app's own files, its options and its configuration folder
  /opt/dashy-app/** r,
  /data/options.json r,
  /config/{,**} rw,

  # run.sh writes the login switch for nginx; nginx its pid, temp files and
  # log, and its error log goes to /dev/stderr
  /run/{,**} rwk,
  /var/lib/nginx/{,**} rw,
  /var/log/nginx/{,**} rw,
  /tmp/{,**} rwk,
  @{PROC}/[0-9]*/fd/[12] w,

  profile node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Status checks, the CORS proxy and the update check reach out over the
    # network. Ping checks send ICMP echo requests with a datagram socket, or
    # with a raw socket where the host does not allow those; ping is setuid
    # root, Node.js itself runs without the capability.
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network inet raw,
    network inet6 raw,
    network unix stream,
    capability net_raw,

    signal (receive) peer=unconfined,
    signal (receive) peer=runc,
    signal (receive) peer=*_dashy,
    signal (send, receive) peer=*_dashy//node,

    # Programs: Node.js itself, and ping for the ping check
    /usr/local/bin/node rix,
    /bin/ping rix,

    # System files and libraries
    /etc/** r,
    /lib/** mr,
    /usr/lib/** mr,
    /usr/share/zoneinfo/** r,

    # Dashy, the login check, and the configuration folder Dashy saves to
    /app/** r,
    /opt/dashy-app/** r,
    /config/{,**} rw,
    /tmp/{,**} rwk,

    # Node.js reads its memory limit and the system load
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    @{PROC}/{cpuinfo,loadavg,meminfo,stat,uptime,version,version_signature} r,
    @{PROC}/[0-9]*/{cgroup,maps,mountinfo,stat,status} r,
  }
}
