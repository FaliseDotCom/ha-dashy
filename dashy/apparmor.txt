#include <tunables/global>

# AppArmor profile for the Dashy app. The Supervisor loads it instead of
# Docker's default profile and renames the profile on the next line to the
# app's full slug, such as local_dashy or 1a2b3c4d_dashy; the signal rules
# match that name with *_dashy. Only the top profile may start at the
# beginning of a line: the Supervisor refuses a file with more than one.
#
# The top profile covers tini, run.sh and the tools it runs as root, and
# nginx. Node.js, so Dashy, the login check and the health check, runs in the
# child profile "node", which can read its own files, write only to the app
# configuration folder and /tmp, and run nothing but ping.
#
# .docs/development.md explains how to test a change to this file.
profile dashy flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # run.sh gives /config to the node user and starts Dashy and the login check
  # as node with su-exec; the nginx master starts its worker as nginx.
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  network inet stream,
  network inet6 stream,
  network unix stream,

  # Docker and the Supervisor send stop signals from outside the profile.
  signal (receive) peer=unconfined,
  signal (send, receive) peer=*_dashy,
  signal (send) peer=*_dashy//node,

  # Programs
  /sbin/tini ix,
  /run.sh r,
  /bin/bash ix,
  /bin/busybox ix,
  /usr/bin/jq ix,
  /sbin/su-exec ix,
  /usr/sbin/nginx ix,
  /usr/local/bin/node cx -> node,

  # System files and libraries
  /etc/** r,
  /lib/** mr,
  /usr/lib/** mr,
  /usr/share/** r,

  # The app's own files, its options and its configuration folder
  /opt/dashy-app/** r,
  /data/options.json r,
  /config/{,**} rw,

  # run.sh writes the login switch for nginx; nginx its pid, temp files and log
  /run/{,**} rwk,
  /var/lib/nginx/{,**} rw,
  /var/log/nginx/{,**} rw,
  /tmp/{,**} rwk,

  profile node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    # Status checks, the CORS proxy and the update check reach out over the
    # network; ping checks send ICMP echo requests.
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,

    signal (receive) peer=unconfined,
    signal (receive) peer=*_dashy,
    signal (send, receive) peer=*_dashy//node,

    # Programs: Node.js itself, and ping for the ping check
    /usr/local/bin/node rix,
    /bin/ping rix,

    # System files and libraries
    /etc/** r,
    /lib/** mr,
    /usr/lib/** mr,
    /usr/share/zoneinfo/** r,

    # Dashy, the login check, and the configuration folder Dashy saves to
    /app/** r,
    /opt/dashy-app/** r,
    /config/{,**} rw,
    /tmp/{,**} rwk,

    # Node.js reads its memory limit and the system load
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
    @{PROC}/{cpuinfo,loadavg,meminfo,stat,uptime,version} r,
    @{PROC}/@{pid}/{cgroup,maps,mountinfo,stat,status} r,
  }
}
